// app/checkout/page.tsx
'use client';
import React, { useState, useEffect } from 'react';
import { useStore } from '@/store/useStore';
import { useRouter } from 'next/navigation';

interface DeliveryZone {
  id: string;
  name: string;
  price: number;
  is_expedition: boolean;
}

const DEFAULT_ZONES: DeliveryZone[] = [
  { id: '1', name: 'Cotonou (Intra-muros)', price: 1000, is_expedition: false },
  { id: '2', name: 'Abomey-Calavi', price: 1500, is_expedition: false },
  { id: '3', name: 'Porto-Novo', price: 2000, is_expedition: false },
  { id: '4', name: 'Autres villes du Bénin (Expédition Parakou, Bohicon, etc.)', price: 3500, is_expedition: true },
];

export default function CheckoutPage() {
  const { cart, getCartSubtotal, clearCart } = useStore();
  const router = useRouter();
  const [zones] = useState<DeliveryZone[]>(DEFAULT_ZONES);
  const [selectedZoneId, setSelectedZoneId] = useState<string>(DEFAULT_ZONES[0].id);

  const [formData, setFormData] = useState({
    name: '',
    phone: '',
    city: 'Cotonou',
    address: '',
    notes: '',
  });
  const [isSubmitting, setIsSubmitting] = useState(false);

  const subtotal = getCartSubtotal();
  const selectedZone = zones.find((z) => z.id === selectedZoneId) || zones[0];
  const total = subtotal + Number(selectedZone.price);

  if (cart.length === 0) {
    return (
      <div className="min-h-screen bg-black text-white flex flex-col items-center justify-center p-4">
        <h1 className="text-2xl font-serif text-amber-400 mb-4">Votre panier est vide</h1>
        <button onClick={() => router.push('/boutique')} className="px-6 py-2.5 bg-amber-500 text-black font-bold rounded-lg">
          Accéder à la boutique
        </button>
      </div>
    );
  }

  const handleOrder = async (e: React.FormEvent) => {
    e.preventDefault();
    setIsSubmitting(true);

    // Simulation de validation & enregistrement commande
    setTimeout(() => {
      clearCart();
      alert('🎉 Votre commande a été enregistrée avec succès ! Notre service livraison vous contactera sous peu pour confirmation.');
      router.push('/');
    }, 1200);
  };

  return (
    <div className="min-h-screen bg-black text-zinc-200 py-12 px-4 max-w-6xl mx-auto">
      <h1 className="text-3xl font-serif font-bold text-transparent bg-clip-text bg-gradient-to-r from-amber-200 to-yellow-500 mb-8 text-center">
        Finalisation de votre Commande Royale
      </h1>

      <div className="grid grid-cols-1 lg:grid-cols-12 gap-8">
        {/* Formulaire */}
        <form onSubmit={handleOrder} className="lg:col-span-7 bg-zinc-900/80 border border-zinc-800 p-6 rounded-2xl space-y-4">
          <h2 className="text-xl font-bold text-amber-400 mb-4">Coordonnées de Livraison</h2>
          
          <div>
            <label className="block text-xs uppercase tracking-wider text-zinc-400 mb-1">Nom et Prénom *</label>
            <input
              required
              type="text"
              value={formData.name}
              onChange={(e) => setFormData({ ...formData, name: e.target.value })}
              className="w-full bg-zinc-950 border border-zinc-700 rounded-lg p-3 text-white focus:border-amber-500 outline-none"
              placeholder="Ex: Jean DOSSOU"
            />
          </div>

          <div>
            <label className="block text-xs uppercase tracking-wider text-zinc-400 mb-1">Numéro de Téléphone (WhatsApp actif) *</label>
            <input
              required
              type="tel"
              value={formData.phone}
              onChange={(e) => setFormData({ ...formData, phone: e.target.value })}
              className="w-full bg-zinc-950 border border-zinc-700 rounded-lg p-3 text-white focus:border-amber-500 outline-none"
              placeholder="Ex: +229 97 00 00 00"
            />
          </div>

          <div>
            <label className="block text-xs uppercase tracking-wider text-zinc-400 mb-1">Zone de Destination *</label>
            <select
              value={selectedZoneId}
              onChange={(e) => setSelectedZoneId(e.target.value)}
              className="w-full bg-zinc-950 border border-zinc-700 rounded-lg p-3 text-white focus:border-amber-500 outline-none"
            >
              {zones.map((zone) => (
                <option key={zone.id} value={zone.id}>
                  {zone.name} (+{zone.price.toLocaleString('fr-FR')} FCFA)
                </option>
              ))}
            </select>
          </div>

          {selectedZone.is_expedition && (
            <div className="bg-amber-950/40 border border-amber-500/40 p-3 rounded-lg text-amber-300 text-xs">
              ⚠️ <strong>Livraison en dehors de Cotonou, Calavi et Porto-Novo :</strong> votre colis sera expédié via notre réseau de transport partenaire après validation téléphonique.
            </div>
          )}

          <div>
            <label className="block text-xs uppercase tracking-wider text-zinc-400 mb-1">Quartier & Repère Précis *</label>
            <textarea
              required
              rows={3}
              value={formData.address}
              onChange={(e) => setFormData({ ...formData, address: e.target.value })}
              className="w-full bg-zinc-950 border border-zinc-700 rounded-lg p-3 text-white focus:border-amber-500 outline-none"
              placeholder="Ex: Haie Vive, 2ème rue après la pharmacie, maison portail vert"
            />
          </div>

          <div className="p-4 bg-zinc-950 border border-emerald-500/30 rounded-xl">
            <div className="text-sm font-semibold text-emerald-400 flex items-center gap-2">
              <span>💵 Mode de paiement unique :</span>
              <span>Paiement à la livraison (Cash on Delivery)</span>
            </div>
            <p className="text-xs text-zinc-500 mt-1">Vous réglez en espèces ou Mobile Money lors de la réception de vos boissons.</p>
          </div>

          <button
            type="submit"
            disabled={isSubmitting}
            className="w-full py-4 bg-gradient-to-r from-amber-500 to-yellow-600 hover:from-amber-600 hover:to-yellow-700 text-black font-bold uppercase tracking-wider rounded-xl transition shadow-lg mt-6"
          >
            {isSubmitting ? 'Traitement en cours...' : `Confirmer ma commande (${total.toLocaleString('fr-FR')} FCFA)`}
          </button>
        </form>

        {/* Récapitulatif Panier */}
        <div className="lg:col-span-5 bg-zinc-900/80 border border-zinc-800 p-6 rounded-2xl flex flex-col justify-between">
          <div>
            <h2 className="text-xl font-bold text-white mb-4">Récapitulatif Panier</h2>
            <div className="divide-y divide-zinc-800 max-h-80 overflow-y-auto pr-2">
              {cart.map((item) => (
                <div key={item.product.id} className="py-3 flex justify-between items-center text-sm">
                  <div>
                    <p className="font-semibold text-white">{item.product.name}</p>
                    <p className="text-xs text-zinc-400">
                      {item.quantity} x {item.unitPrice.toLocaleString('fr-FR')} FCFA 
                      {item.isWholesale && <span className="text-amber-400 ml-1 font-bold">(Tarif Gros)</span>}
                    </p>
                  </div>
                  <span className="font-mono text-amber-300 font-bold">
                    {(item.quantity * item.unitPrice).toLocaleString('fr-FR')} FCFA
                  </span>
                </div>
              ))}
            </div>
          </div>

          <div className="border-t border-zinc-800 pt-4 mt-6 space-y-2 text-sm">
            <div className="flex justify-between text-zinc-400">
              <span>Sous-total</span>
              <span>{subtotal.toLocaleString('fr-FR')} FCFA</span>
            </div>
            <div className="flex justify-between text-zinc-400">
              <span>Frais de livraison ({selectedZone.name})</span>
              <span>{Number(selectedZone.price).toLocaleString('fr-FR')} FCFA</span>
            </div>
            <div className="flex justify-between text-lg font-bold text-white border-t border-zinc-700 pt-3">
              <span>Total Général à régler</span>
              <span className="text-amber-400">{total.toLocaleString('fr-FR')} FCFA</span>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
}
